### For All Components
```jsx

function MyComponent() {
  const { isAuth, user, login, logout } = useAuth();
  // No more prop drilling!
}
```

### For API Calls
```jsx
import { normalizeResponse, extractToken } from '../utils/apiHelpers';

const response = await fetchCourses();
const courses = normalizeResponse(response); // Handles any shape
```

### For Error Handling
```jsx
import { getUserErrorMessage } from '../utils/errorHandler';

try {
  await updateAccount(data);
} catch (error) {
  const message = getUserErrorMessage(error); // User-friendly
  showAlert(message);
}
```

### For Caching
```jsx
import { getCachedCourses, setCachedCourses } from '../utils/storage';

const cached = getCachedCourses(); // Replaces localStorage
setCachedCourses(newCourses);
```
